│ 💡 INTEGRATION PATTERNS                                                   │

  ┌─────────────────────────────────────────────────────────────────────────┐
  │ Pattern 1: Login Success/Failure                                        │
  └─────────────────────────────────────────────────────────────────────────┘

	Private Sub btnLogin_Click(sender As Object, e As EventArgs) _
							   Handles btnLogin.Click
		Dim user As UserAccount = ModuleDatabase.AuthenticateUser(
			txtUsername.Text, txtPassword.Text)

		If user IsNot Nothing Then
			' SUCCESS
			ModuleDatabase.LogSystemActivity(
				user.Username,
				"User Login - Successful",
				"FormLogin"
			)
		Else
			' FAILURE
			ModuleDatabase.LogSystemActivity(
				txtUsername.Text,
				"User Login - Failed Authentication",
				"FormLogin",
				severity:="WARNING"
			)
		End If
	End Sub

  ┌─────────────────────────────────────────────────────────────────────────┐
  │ Pattern 2: Data Modification with Context                               │
  └─────────────────────────────────────────────────────────────────────────┘

	Private Sub btnSave_Click(sender As Object, e As EventArgs) _
							  Handles btnSave.Click
		Try
			Dim patient As New PatientModel With {
				.PatientID = txtPatientID.Text
			}

			If ModuleDatabase.SavePatient(patient) Then
				ModuleDatabase.LogSystemActivity(
					username:=SessionManager.CurrentUser.Username,
					action:="Patient Record Saved",
					moduleName:="FormPatientManagement",
					severity:="INFO",
					additionalContext:=$"{{""PatientID"":""{patient.PatientID}""}}"
				)
			End If

		Catch ex As Exception
			ModuleDatabase.LogSystemActivity(
				username:=SessionManager.CurrentUser.Username,
				action:="Patient Save Failed",
				moduleName:="FormPatientManagement",
				severity:="ERROR",
				additionalContext:=ex.Message
			)
		End Try
	End Sub

  ┌─────────────────────────────────────────────────────────────────────────┐
  │ Pattern 3: Access Control Validation                                    │
  └─────────────────────────────────────────────────────────────────────────┘

	Private Sub FormDoctorsManagement_Load(sender As Object, e As EventArgs) _
										   Handles MyBase.Load
		If SessionManager.CurrentUser Is Nothing OrElse _
		   SessionManager.CurrentUser.Role <> "Admin" Then

			' CRITICAL: Unauthorized access attempt
			ModuleDatabase.LogSystemActivity(
				username:=If(SessionManager.CurrentUser IsNot Nothing,
						   SessionManager.CurrentUser.Username, "UNKNOWN"),
				action:="Unauthorized Access Attempt - Doctors Management",
				moduleName:="FormDoctorsManagement",
				severity:="CRITICAL",
				additionalContext:="Non-admin user attempted admin module"
			)

			MessageBox.Show("Access Denied.")
			Me.Close()
			Return
		End If

		' SUCCESS: Authorized access
		ModuleDatabase.LogSystemActivity(
			SessionManager.CurrentUser.Username,
			"Doctors Management Accessed",
			"FormDoctorsManagement"
		)
	End Sub

┌───────────────────────────────────────────────────────────────────────────┐
│ ⚠️ IMPORTANT NOTES                                                        │
└───────────────────────────────────────────────────────────────────────────┘

  ✓ ALWAYS use SessionManager.CurrentUser.Username for accurate tracking
  ✓ Use descriptive action strings (not just "Update" or "Delete")
  ✓ Include context for data modifications (PatientID, RecordID, etc.)
  ✓ Log BOTH success AND failure scenarios
  ✓ Use appropriate severity levels
  ✓ Emergency backup is automatic (no manual intervention needed)
  ✓ Thread-safe (safe for concurrent logging)
  ✓ Parameterized queries (SQL injection protected)

┌───────────────────────────────────────────────────────────────────────────┐
│ 🔍 TROUBLESHOOTING                                                        │
└───────────────────────────────────────────────────────────────────────────┘

  Problem: Logs not appearing in database
  ────────────────────────────────────────
  1. Check audit_emergency_backup.txt for failover records
  2. Check error_log.txt for exceptions
  3. Verify HospitalDB.db is writable
  4. Ensure InitialiseDatabase() was called on startup

  Problem: Query returns no results
  ──────────────────────────────────
  1. Verify filters are correct (case-sensitive for severity)
  2. Check date format (yyyy-MM-dd)
  3. Use GetAuditLogs() with no parameters to get all logs
