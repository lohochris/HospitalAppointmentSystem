#Region "Telehealth Button - Programmatic Creation"
	''' <summary>
	''' Programmatically creates the Launch Telehealth button and wires event handler.
	''' Call this from FormMain_Load or after InitializeComponent().
	''' </summary>
	Private Sub InitializeTelehealthButton()
		Try
			' Create button with medical-grade styling
			btnLaunchTelehealth = New Button With {
				.Name = "btnLaunchTelehealth",
				.Text = "📞 Launch Telehealth",
				.Size = New Size(120, 35),
				.Location = New Point(20, 420),  ' Adjust Y coordinate based on your layout
				.BackColor = Color.FromArgb(0, 120, 212),  ' Medical Blue
				.ForeColor = Color.White,
				.FlatStyle = FlatStyle.Flat,
				.Font = New Font("Arial", 10, FontStyle.Bold),
				.Cursor = Cursors.Hand,
				.TabIndex = 7,
				.UseVisualStyleBackColor = False,
				.Anchor = AnchorStyles.Top Or AnchorStyles.Left
			}

			' Modern flat appearance
			btnLaunchTelehealth.FlatAppearance.BorderSize = 0
			btnLaunchTelehealth.FlatAppearance.MouseOverBackColor = Color.FromArgb(0, 130, 220)
			btnLaunchTelehealth.FlatAppearance.MouseDownBackColor = Color.FromArgb(0, 100, 180)

			' Wire event handler (implementation already exists in FormMain.vb)
			AddHandler btnLaunchTelehealth.Click, AddressOf btnLaunchTelehealth_Click

			' Add to form
			Me.Controls.Add(btnLaunchTelehealth)
			btnLaunchTelehealth.BringToFront()

			ModuleDatabase.LogError("InitializeTelehealthButton: Button created and wired successfully")

		Catch ex As Exception
			ModuleDatabase.LogError($"InitializeTelehealthButton error: {ex.Message}")
			MessageBox.Show($"Failed to create telehealth button: {ex.Message}", 
						   "Initialization Error", MessageBoxButtons.OK, MessageBoxIcon.Warning)
		End Try
	End Sub
#End Region

#Region "Optional: Keyboard Shortcut (Ctrl+T)"
	''' <summary>
	''' Adds Ctrl+T keyboard shortcut to launch telehealth.
	''' Also set: Me.KeyPreview = True in Form properties or Form_Load
	''' </summary>
	Private Sub FormMain_KeyDown(sender As Object, e As KeyEventArgs) Handles Me.KeyDown
		Try
			' Ctrl+T: Quick Launch Telehealth
			If e.Control AndAlso e.KeyCode = Keys.T Then
				e.Handled = True
				e.SuppressKeyPress = True
				btnLaunchTelehealth_Click(Me, EventArgs.Empty)
			End If
		Catch ex As Exception
			ModuleDatabase.LogError($"FormMain_KeyDown error: {ex.Message}")
		End Try
	End Sub
#End Region

#Region "Optional: Double-Click Row to Launch"
	''' <summary>
	''' Launches telehealth when user double-clicks an appointment row.
	''' </summary>
	Private Sub dgvDashboardData_CellDoubleClick(sender As Object, e As DataGridViewCellEventArgs) Handles dgvDashboardData.CellDoubleClick
		Try
			' Ignore header row double-clicks
			If e.RowIndex < 0 Then Return

			' Auto-launch telehealth for double-clicked row
			btnLaunchTelehealth_Click(sender, EventArgs.Empty)

		Catch ex As Exception
			ModuleDatabase.LogError($"dgvDashboardData_CellDoubleClick error: {ex.Message}")
		End Try
	End Sub
#End Region
